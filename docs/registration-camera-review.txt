async function handlePhotoSelection(input) {
  const file = input.files?.[0];
  if (!file) return;
  camera.close();
  if (photoPreview.src.startsWith('blob:')) URL.revokeObjectURL(photoPreview.src);
  form.hidden = true;
  photoDataUrl = '';
  closetImageDataUrl = '';
  analyzeButton.disabled = true;
  analysisStatus.textContent = '';
  photoPreview.hidden = true;
  if (!file.type.startsWith('image/')) {
    showStatus(analysisStatus, '사진 파일을 골라 주세요.', true);
    return;
  }
  try {
    photoDataUrl = await readRegistrationPhoto(file);
    photoPreview.src = URL.createObjectURL(file);
    photoPreview.hidden = false;
    analyzeButton.disabled = !garmentTypeInput.value;
    showStatus(analysisStatus, '사진을 골랐어요. 다음 버튼을 눌러 옷을 찾아봐요.');
  } catch (error) {
    showStatus(analysisStatus, error.message, true);
  }
}

photoInputs.forEach((input) => input.addEventListener('change', () => handlePhotoSelection(input)));

analyzeButton.addEventListener('click', runAnalysis);

garmentTypeInput.addEventListener('change', () => {
  analyzeButton.disabled = !photoDataUrl || !garmentTypeInput.value;

async function runAnalysis() {
  analyzeButton.disabled = true;
  form.hidden = true;
  showStatus(analysisStatus, '고른 배경의 표시와 옷 모양을 찾고 있어요. 잠시 기다려 주세요.');
  try {
    const result = await requestJson('/api/garments/analyze', {
      method: 'POST', body: JSON.stringify({ backgroundId: backgroundInput.value, garmentId: Number(garmentTypeInput.value), photoDataUrl })
    });
    fillForm(result);
    form.hidden = false;
    showStatus(analysisStatus, result.message);
    form.scrollIntoView({ behavior: 'smooth', block: 'start' });
  } catch (error) {
    showStatus(analysisStatus, error.message, true);
  } finally {
    analyzeButton.disabled = false;
  }
}

form.addEventListener('submit', async (event) => {
  event.preventDefault();
  if (!photoDataUrl) return showStatus(saveStatus, '저장할 사진을 먼저 골라 주세요.', true);
  const values = new FormData(form);
  const garmentTypeId = Number(garmentTypeInput.value);
  const payload = {
    name: values.get('name'), category: categoryForId(garmentTypeId), garmentTypeId, gender: values.get('gender'),
    lengthCm: Number(values.get('lengthCm')), widthCm: Number(values.get('widthCm')),
    colorName: values.get('colorName'), photoDataUrl, closetImageDataUrl, backgroundId: backgroundInput.value
  };
  const saveButton = form.querySelector('button[type="submit"]');
  saveButton.disabled = true;
  showStatus(saveStatus, '옷과 사진을 서버에 저장하고 있어요.');
  try {
    const saved = await requestJson('/api/garments', { method: 'POST', body: JSON.stringify(payload) });
    showStatus(analysisStatus, `${saved.name}을(를) 옷장에 저장했어요!`);
    await loadGarments();
    form.reset();
    form.hidden = true;
    photoInputs.forEach((input) => { input.value = ''; });
    photoDataUrl = '';
    closetImageDataUrl = '';
    backgroundInput.value = 'M4';
    garmentTypeInput.value = '';
    photoPreview.hidden = true;
    analyzeButton.disabled = true;
  } catch (error) {
    showStatus(saveStatus, error.message, true);
  } finally {
    saveButton.disabled = false;
  }
});

function fillForm(result) {
  form.elements.lengthCm.value = result.lengthCm;
  form.elements.widthCm.value = result.widthCm;
  form.elements.colorName.value = result.colorName;
  form.elements.name.value = '';
  document.querySelector('#width-label').textContent = `${result.widthLabel}(cm)`;
  document.querySelector('#chosen-garment-type').textContent = `선택한 종류: ${result.garmentName}`;
  document.querySelector('#analysis-preview').src = result.analysisPhotoDataUrl;
  closetImageDataUrl = result.closetImageDataUrl;

}

async function loadGarments() {
  return garmentClosetView.reload();

function categoryForId(garmentId) {
  if (garmentId === 3) return 'dress';
  if (garmentId === 5) return 'pants';
  return 'top';
}

async function requestJson(url, options = {}) {
  let response;
  try {
    response = await fetch(url, {
      ...options,
      headers: { 'Content-Type': 'application/json', ...options.headers }
    });
  } catch {
    throw new Error('저장 서버에 연결할 수 없어요. FitShare 서버가 켜져 있는지 확인해 주세요.');
  }
  const result = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(result.error || '요청을 마치지 못했어요. 사진을 확인하고 다시 해 주세요.');
  return result;
}

// 촬영한 JPEG를 기존 사진 선택 흐름에 넘깁니다. 분석·저장 방식은 바꾸지 않습니다.
export function createRegistrationCamera(ui, browser = {}) {
  const mediaDevices = browser.mediaDevices ?? globalThis.navigator?.mediaDevices;
  const secureContext = browser.secureContext ?? globalThis.isSecureContext;
  const documentRef = browser.documentRef ?? document;
  const windowRef = browser.windowRef ?? window;
  const FileClass = browser.FileClass ?? globalThis.File;
  let stream = null;
  let session = 0;

  function showStatus(message, isError = false) {
    ui.status.textContent = message;
    ui.status.classList.toggle('is-error', isError);
  }

  function close() {
    session += 1;
    stream?.getTracks().forEach((track) => track.stop());
    stream = null;
    ui.video.pause();
    ui.video.srcObject = null;
    ui.panel.hidden = true;
    ui.captureButton.disabled = true;
    ui.openButton.disabled = false;
  }

  function ready() {
    ui.captureButton.disabled = !stream || !ui.video.videoWidth || !ui.video.videoHeight;
  }

  async function open() {
    close();
    const currentSession = session;
    ui.panel.hidden = false;
    showStatus('카메라를 켜고 있어요. 카메라 사용 요청이 뜨면 허용해 주세요.');
    ui.cancelButton.focus();
    if (!secureContext) {
      showStatus('노트북에서는 http://localhost:8000/measure/register.html 로 열어 주세요. 다른 기기에서 직접 촬영하려면 HTTPS 연결이 필요해요.', true);
      return;
    }
    if (!mediaDevices?.getUserMedia) {
      showStatus('이 브라우저는 직접 촬영을 지원하지 않아요. 최신 Chrome이나 Edge를 쓰거나 갤러리에서 사진을 골라 주세요.', true);
      return;
    }
    ui.openButton.disabled = true;
    try {
      const openedStream = await mediaDevices.getUserMedia({
        audio: false,
        video: { facingMode: { ideal: 'environment' }, width: { ideal: 1920 }, height: { ideal: 1080 } }
      });
      // 권한 요청 중 닫았다면, 늦게 켜진 카메라도 바로 끕니다.
      if (session !== currentSession) {
        openedStream.getTracks().forEach((track) => track.stop());
        return;
      }
      stream = openedStream;
      ui.video.srcObject = stream;
      stream.getTracks().forEach((track) => track.addEventListener('ended', () => {
        if (session !== currentSession) return;
        close();
        ui.panel.hidden = false;
        showStatus('카메라 연결이 끊겼어요. 카메라로 촬영 버튼을 다시 눌러 주세요.', true);
      }));
      await ui.video.play();
      if (session !== currentSession) return;
      ready();
      showStatus('옷 전체와 매트가 화면에 보이면 사진 찍기를 눌러 주세요.');
    } catch (error) {
      if (session !== currentSession) return;
      close();
      ui.panel.hidden = false;
      const messages = {
        NotAllowedError: '카메라 권한이 꺼져 있어요. 주소창 옆 사이트 설정에서 카메라를 허용한 뒤 다시 눌러 주세요.',
        NotFoundError: '사용할 카메라를 찾지 못했어요. 노트북 카메라가 연결되어 있는지 확인해 주세요.',
        NotReadableError: '카메라를 켜지 못했어요. 다른 앱이 사용 중인지, 카메라 차단 스위치가 켜져 있는지 확인해 주세요.'
      };
      showStatus(messages[error.name] || '카메라를 켜지 못했어요. 연결과 권한을 확인하고 다시 시도해 주세요.', true);
    }
  }

  async function capture() {
    if (!stream || ui.captureButton.disabled || !ui.video.videoWidth || !ui.video.videoHeight) return;
    const currentSession = session;
    ui.captureButton.disabled = true;
    try {
      const canvas = documentRef.createElement('canvas');
      canvas.width = ui.video.videoWidth;
      canvas.height = ui.video.videoHeight;
      const context = canvas.getContext('2d');
      if (!context) throw new Error('사진을 만들지 못했어요. 다시 촬영해 주세요.');
      context.drawImage(ui.video, 0, 0, canvas.width, canvas.height);
      const blob = await new Promise((resolve) => canvas.toBlob(resolve, 'image/jpeg', 1));
      if (session !== currentSession) return;
      if (!blob) throw new Error('사진을 만들지 못했어요. 다시 촬영해 주세요.');
      const file = new FileClass([blob], 'camera-photo.jpg', { type: 'image/jpeg' });
      close();
      await ui.onPhoto(file);
      ui.openButton.focus();
    } catch (error) {
      if (session !== currentSession) return;
      showStatus(error.message || '사진을 찍지 못했어요. 다시 촬영해 주세요.', true);
      ready();
    }
  }

  ui.openButton.addEventListener('click', open);
  ui.captureButton.addEventListener('click', capture);
  ui.cancelButton.addEventListener('click', () => { close(); ui.openButton.focus(); });
  ui.video.addEventListener('loadeddata', ready);
  windowRef.addEventListener('pagehide', close);
  windowRef.addEventListener('keydown', (event) => {
    if (event.key === 'Escape' && !ui.panel.hidden) { close(); ui.openButton.focus(); }
  });
  return { open, close, capture };
}

import assert from 'node:assert/strict';
import { Blob, File } from 'node:buffer';

export async function runCameraTests(createRegistrationCamera) {
  const results = [];
  function fixture(getUserMedia, secureContext = true) {
    const element = () => ({ hidden: true, disabled: false, textContent: '',
      listeners: {}, addEventListener(name, fn) { this.listeners[name] = fn; },
      classList: { toggle() {} }, focus() {} });
    const video = { ...element(), videoWidth: 1920, videoHeight: 1080,
      srcObject: null, async play() {}, pause() {} };
    const canvas = { getContext: () => ({ drawImage() {} }),
      toBlob(callback) { callback(new Blob(['photo'], { type: 'image/jpeg' })); } };
    const ui = { openButton: element(), panel: element(), video,
      captureButton: element(), cancelButton: element(), status: element(),
      onPhoto: async (file) => { ui.photo = file; } };
    const windowListeners = {};
    const camera = createRegistrationCamera(ui, { secureContext,
      FileClass: File,
      mediaDevices: { getUserMedia }, documentRef: { createElement: () => canvas },
      windowRef: { addEventListener(name, action) { windowListeners[name] = action; } } });
    return { ui, camera, canvas, windowListeners };
  }
  async function test(name, action) { await action(); results.push(name); }
  const makeStream = () => {
    const track = { stopped: false, stop() { this.stopped = true; }, addEventListener() {} };
    return { track, getTracks: () => [track] };
  };

  await test('정상: 카메라 영상의 원래 크기로 JPEG 촬영하고 카메라 종료', async () => {
    const stream = makeStream();
    let constraints;
    const { ui, camera, canvas } = fixture(async (value) => { constraints = value; return stream; });
    await camera.open();
    assert.equal(ui.panel.hidden, false);
    assert.equal(ui.video.srcObject, stream);
    assert.equal(constraints.audio, false);
    assert.equal(constraints.video.facingMode.ideal, 'environment');
    assert.equal(ui.captureButton.disabled, false);
    await camera.capture();
    assert.equal(ui.photo.type, 'image/jpeg');
    assert.equal(canvas.width, 1920);
    assert.equal(canvas.height, 1080);
    assert.equal(stream.track.stopped, true);
    assert.equal(ui.panel.hidden, true);
  });
  await test('경계: 영상 크기가 아직 없으면 촬영하지 않음', async () => {
    const { ui, camera } = fixture(async () => makeStream());
    ui.video.videoWidth = 0;
    await camera.open();
    await camera.capture();
    assert.equal(ui.photo, undefined);
    assert.equal(ui.captureButton.disabled, true);
    camera.close();
  });
  await test('경계: 카메라 닫기 버튼과 페이지 이동에서 카메라 종료', async () => {
    const stream = makeStream();
    const { ui, camera, windowListeners } = fixture(async () => stream);
    await camera.open();
    ui.cancelButton.listeners.click();
    assert.equal(stream.track.stopped, true);
    assert.equal(ui.video.srcObject, null);
    assert.equal(ui.panel.hidden, true);
    stream.track.stopped = false;
    await camera.open();
    windowListeners.pagehide();
    assert.equal(stream.track.stopped, true);
    assert.equal(ui.video.srcObject, null);
  });
  await test('오류: 영상 재생 실패 시 켜진 카메라도 종료', async () => {
    const stream = makeStream();
    const { ui, camera } = fixture(async () => stream);
    ui.video.play = async () => { throw new Error('play failed'); };
    await camera.open();
    assert.equal(stream.track.stopped, true);
    assert.equal(ui.video.srcObject, null);
    assert.equal(ui.captureButton.disabled, true);
    assert.match(ui.status.textContent, /카메라/);
  });
  await test('오류: 권한 거부 안내 후 다시 시도 가능', async () => {
    const { ui, camera } = fixture(async () => { throw { name: 'NotAllowedError' }; });
    await camera.open();
    assert.match(ui.status.textContent, /권한/);
    assert.equal(ui.openButton.disabled, false);
    assert.equal(ui.captureButton.disabled, true);
  });
  await test('오류: 카메라 없음 안내', async () => {
    const { ui, camera } = fixture(async () => { throw { name: 'NotFoundError' }; });
    await camera.open();
    assert.match(ui.status.textContent, /카메라.*찾/);
  });
  await test('오류: HTTP IP 주소에서는 localhost 안내', async () => {
    let requested = false;
    const { ui, camera } = fixture(async () => { requested = true; }, false);
    await camera.open();
    assert.equal(requested, false);
    assert.match(ui.status.textContent, /localhost/);
  });
  await test('오류: 카메라 API 없는 브라우저도 앱이 깨지지 않음', async () => {
    const { ui, camera } = fixture(undefined);
    await camera.open();
    assert.match(ui.status.textContent, /브라우저/);
    assert.equal(ui.captureButton.disabled, true);
  });
  await test('경계: 권한 응답 전에 취소해도 늦게 도착한 카메라 종료', async () => {
    let resolve;
    const stream = makeStream();
    const { ui, camera } = fixture(() => new Promise((done) => { resolve = done; }));
    const opening = camera.open();
    camera.close();
    resolve(stream);
    await opening;
    assert.equal(stream.track.stopped, true);
    assert.equal(ui.video.srcObject, null);
    assert.equal(ui.panel.hidden, true);
  });
  await test('오류: JPEG 변환 실패해도 다시 촬영 가능', async () => {
    const { ui, camera, canvas } = fixture(async () => makeStream());
    canvas.toBlob = (callback) => callback(null);
    await camera.open();
    await camera.capture();
    assert.match(ui.status.textContent, /사진/);
    assert.equal(ui.captureButton.disabled, false);
    assert.equal(ui.photo, undefined);
    camera.close();
  });
  await test('경계: JPEG 변환 중 닫으면 늦게 완성된 사진을 선택하지 않음', async () => {
    const { ui, camera, canvas } = fixture(async () => makeStream());
    let finish;
    canvas.toBlob = (callback) => { finish = callback; };
    await camera.open();
    const capture = camera.capture();
    camera.close();
    finish(new Blob(['photo'], { type: 'image/jpeg' }));
    await capture;
    assert.equal(ui.photo, undefined);
    assert.equal(ui.panel.hidden, true);
  });
  return results;
}
